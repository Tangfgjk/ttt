# 类设计（面向对象详细设计）

## 1. 设计原则

- **SOLID 原则**：单一职责、开闭原则、里氏替换、接口隔离、依赖倒置
- **策略模式**：Coreset 算法族可插拔替换
- **工厂模式**：算法注册与实例化统一入口
- **观察者模式**：标注事件驱动嵌入更新、推荐重算、模型重训练

### 3.1 抽象基类：Coreset 选择器

```python
from abc import ABC, abstractmethod
from typing import List, Tuple, Optional
import numpy as np


class CoresetSelectionResult:
    """Coreset 选择结果的数据容器"""
    def __init__(self, 
                 selected_indices: np.ndarray,
                 scores: np.ndarray,
                 algorithm_name: str,
                 metadata: dict = None):
        self.selected_indices = selected_indices  # 选中的样本在池中的索引
        self.scores = scores                      # 每个样本的分数
        self.algorithm_name = algorithm_name
        self.metadata = metadata or {}            # 算法特有元信息
        
    def top_k(self, k: int) -> np.ndarray:
        """返回分数最高的 k 个索引"""
        return self.selected_indices[np.argsort(self.scores)[-k:][::-1]]


class BaseCoresetSelector(ABC):
    """
    Coreset 选择算法抽象基类。
    所有具体的 Coreset 算法必须继承此类，实现 select 和 score 方法。
    """
    
    name: str = "base"  # 子类必须覆盖
    
    def __init__(self, params: dict = None):
        self.params = params or {}
        self._is_fitted = False
    
    @abstractmethod
    def select(self,
               pool_embeddings: np.ndarray,
               pool_ids: List[int],
               labeled_embeddings: Optional[np.ndarray] = None,
               labeled_ids: Optional[List[int]] = None,
               model_outputs: Optional[np.ndarray] = None,
               n_select: int = 10) -> CoresetSelectionResult:
        """
        从候选池中选出 n_select 个样本。
        
        Args:
            pool_embeddings: 候选池样本的嵌入矩阵 [N_pool, D]
            pool_ids: 候选池样本的业务ID列表
            labeled_embeddings: 已标注样本的嵌入矩阵 [N_labeled, D]，可选
            labeled_ids: 已标注样本的业务ID列表，可选
            model_outputs: 当前模型对候选池的预测输出 [N_pool, C]，用于不确定性方法
            n_select: 需要选择的样本数
            
        Returns:
            CoresetSelectionResult: 包含选中索引、分数和元信息
        """
        pass
    
    @abstractmethod
    def score(self,
              pool_embeddings: np.ndarray,
              labeled_embeddings: Optional[np.ndarray] = None,
              model_outputs: Optional[np.ndarray] = None) -> np.ndarray:
        """
        为候选池中的每个样本计算选择分数（越高越优先）。
        
        Returns:
            np.ndarray: 分数数组 [N_pool]
        """
        pass
    
    def incremental_update(self,
                           new_labeled_indices: List[int],
                           previous_result: CoresetSelectionResult) -> CoresetSelectionResult:
        """
        增量更新：当有新样本被标注后，更新推荐结果。
        默认实现为全量重算，子类可覆盖以提供更高效的增量逻辑。
        """
        # 默认回退到全量重算
        return previous_result
```

### 3.2 工厂与注册器

```python
class CoresetAlgorithmFactory:
    """Coreset 算法工厂，统一管理所有算法的注册与实例化。"""
    
    _registry: dict[str, type[BaseCoresetSelector]] = {}
    
    @classmethod
    def register(cls, name: str):
        """装饰器：自动注册算法类"""
        def wrapper(selector_cls: type[BaseCoresetSelector]):
            cls._registry[name] = selector_cls
            selector_cls.name = name
            return selector_cls
        return wrapper
    
    @classmethod
    def get(cls, name: str, params: dict = None) -> BaseCoresetSelector:
        if name not in cls._registry:
            raise ValueError(f"Unknown coreset algorithm: {name}. "
                           f"Available: {list(cls._registry.keys())}")
        return cls._registry[name](params)
    
    @classmethod
    def list_algorithms(cls) -> List[str]:
        return sorted(cls._registry.keys())
```

### 3.3 具体算法示例

```python
from sklearn.cluster import KMeans
from scipy.spatial.distance import cdist


@CoresetAlgorithmFactory.register("kmeans")
class KMeansCoresetSelector(BaseCoresetSelector):
    """
    K-Means 几何覆盖法。
    在特征空间运行 K-Means，选择距离聚类中心最近的样本。
    """
    
    def select(self, pool_embeddings, pool_ids, labeled_embeddings=None,
               labeled_ids=None, model_outputs=None, n_select=10):
        kmeans = KMeans(n_clusters=n_select, random_state=42, n_init=10)
        kmeans.fit(pool_embeddings)
        
        # 对每个中心，找最近的样本
        distances = cdist(pool_embeddings, kmeans.cluster_centers_, metric='euclidean')
        selected_indices = np.argmin(distances, axis=0)
        
        # 分数：到最近中心的距离的负值（越近分数越高）
        scores = -np.min(distances, axis=1)
        
        return CoresetSelectionResult(
            selected_indices=selected_indices,
            scores=scores,
            algorithm_name=self.name,
            metadata={"inertia": float(kmeans.inertia_)}
        )
    
    def score(self, pool_embeddings, labeled_embeddings=None, model_outputs=None):
        # 简化版：返回均匀分布的分数，实际可结合聚类密度
        return np.ones(len(pool_embeddings))


@CoresetAlgorithmFactory.register("uncertainty")
class UncertaintyCoresetSelector(BaseCoresetSelector):
    """
    不确定性采样：基于模型输出的熵或 Margin。
    适用于已有初步模型时的主动学习场景。
    """
    
    def __init__(self, params=None):
        super().__init__(params)
        self.strategy = self.params.get("strategy", "entropy")  # entropy / margin / least_confidence
    
    def select(self, pool_embeddings, pool_ids, labeled_embeddings=None,
               labeled_ids=None, model_outputs=None, n_select=10):
        assert model_outputs is not None, "Uncertainty 方法需要 model_outputs"
        scores = self.score(pool_embeddings, model_outputs=model_outputs)
        top_k = np.argsort(scores)[-n_select:][::-1]
        return CoresetSelectionResult(
            selected_indices=top_k,
            scores=scores,
            algorithm_name=self.name,
            metadata={"strategy": self.strategy}
        )
    
    def score(self, pool_embeddings, labeled_embeddings=None, model_outputs=None):
        probs = model_outputs
        if self.strategy == "entropy":
            scores = -np.sum(probs * np.log(probs + 1e-10), axis=1)
        elif self.strategy == "margin":
            sorted_probs = np.sort(probs, axis=1)
            scores = -(sorted_probs[:, -1] - sorted_probs[:, -2])
        elif self.strategy == "least_confidence":
            scores = -np.max(probs, axis=1)
        else:
            raise ValueError(f"Unknown strategy: {self.strategy}")
        return scores


@CoresetAlgorithmFactory.register("herding")
class HerdingCoresetSelector(BaseCoresetSelector):
    """
    Herding 算法：贪心选择使核心集均值逼近全集均值的样本。
    来自 Chen et al. 2010, 在实验数据中 MMD 表现最佳。
    """
    
    def select(self, pool_embeddings, pool_ids, labeled_embeddings=None,
               labeled_ids=None, model_outputs=None, n_select=10):
        n_pool = len(pool_embeddings)
        mean_all = np.mean(pool_embeddings, axis=0)
        selected = []
        remaining = set(range(n_pool))
        current_mean = np.zeros_like(mean_all)
        
        for _ in range(n_select):
            if not remaining:
                break
            # 选择使当前均值最接近目标均值的样本
            candidates = np.array(list(remaining))
            temp_means = (current_mean * len(selected) + pool_embeddings[candidates]) / (len(selected) + 1)
            distances = np.linalg.norm(temp_means - mean_all, axis=1)
            best_idx_in_candidates = np.argmin(distances)
            best_idx = candidates[best_idx_in_candidates]
            
            selected.append(best_idx)
            remaining.remove(best_idx)
            current_mean = temp_means[best_idx_in_candidates]
        
        # 为所有样本打分（已选中为1.0，其余按到已选最近距离反向）
        scores = np.zeros(n_pool)
        if selected:
            dist_to_selected = cdist(pool_embeddings, pool_embeddings[selected], metric='euclidean')
            scores = -np.min(dist_to_selected, axis=1)
            scores[selected] = 1.0
        
        return CoresetSelectionResult(
            selected_indices=np.array(selected),
            scores=scores,
            algorithm_name=self.name
        )
    
    def score(self, pool_embeddings, labeled_embeddings=None, model_outputs=None):
        result = self.select(pool_embeddings, [], n_select=min(10, len(pool_embeddings)))
        return result.scores
```

### 3.4 嵌入服务

```python
from transformers import AutoTokenizer, AutoModel
import torch
from functools import lru_cache


class BaseEmbedder(ABC):
    """文本嵌入器抽象基类"""
    
    @property
    @abstractmethod
    def dimension(self) -> int:
        pass
    
    @abstractmethod
    def encode(self, texts: List[str]) -> np.ndarray:
        """编码文本列表，返回 [N, D]"""
        pass
    
    def encode_batch(self, texts: List[str], batch_size: int = 32) -> np.ndarray:
        """批量编码，带进度条"""
        all_embeddings = []
        for i in range(0, len(texts), batch_size):
            batch = texts[i:i+batch_size]
            all_embeddings.append(self.encode(batch))
        return np.vstack(all_embeddings)


class RoBERTaEmbedder(BaseEmbedder):
    """基于 chinese-roberta-wwm-ext 的嵌入提取"""
    
    def __init__(self, model_name: str = "hfl/chinese-roberta-wwm-ext", device: str = None):
        self.model_name = model_name
        self.device = device or ("cuda" if torch.cuda.is_available() else "cpu")
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name).to(self.device)
        self.model.eval()
    
    @property
    def dimension(self) -> int:
        return self.model.config.hidden_size
    
    @torch.no_grad()
    def encode(self, texts: List[str]) -> np.ndarray:
        inputs = self.tokenizer(texts, return_tensors="pt", padding=True, 
                               truncation=True, max_length=512)
        inputs = {k: v.to(self.device) for k, v in inputs.items()}
        outputs = self.model(**inputs)
        # 使用 [CLS] token 的 pooler output 或 mean pooling
        embeddings = outputs.pooler_output.cpu().numpy()
        return embeddings


class EmbeddingService:
    """
    嵌入计算与缓存服务。
    职责：
    - 管理多个 Embedder 实例
    - 缓存已计算的嵌入 (Redis)
    - 支持全量/增量更新
    """
    
    def __init__(self, embedder: BaseEmbedder, cache_client, db_session):
        self.embedder = embedder
        self.cache = cache_client
        self.db = db_session
    
    def compute_for_question(self, question_id: int, question_text: str, 
                             model_name: str = "roberta") -> np.ndarray:
        """单题嵌入计算，优先读缓存"""
        cache_key = f"emb:{model_name}:{question_id}"
        cached = self.cache.get(cache_key)
        if cached:
            return np.frombuffer(cached, dtype=np.float32)
        
        emb = self.embedder.encode([question_text])[0]
        self.cache.set(cache_key, emb.tobytes(), ex=86400 * 7)
        return emb
    
    def compute_all_embeddings(self, questions: List[dict]) -> Tuple[np.ndarray, List[int]]:
        """批量计算嵌入，返回 (embeddings_matrix, question_ids)"""
        texts = [q["question_text"] for q in questions]
        ids = [q["id"] for q in questions]
        embeddings = self.embedder.encode_batch(texts)
        return embeddings, ids
    
    def compute_tsne(self, embeddings: np.ndarray, perplexity: int = 30) -> np.ndarray:
        from sklearn.manifold import TSNE
        tsne = TSNE(n_components=2, perplexity=perplexity, random_state=42, n_iter=1000)
        return tsne.fit_transform(embeddings)
    
    def compute_umap(self, embeddings: np.ndarray, n_neighbors: int = 15) -> np.ndarray:
        import umap
        reducer = umap.UMAP(n_components=2, n_neighbors=n_neighbors, random_state=42)
        return reducer.fit_transform(embeddings)
```

    """

        4. 调用 Coreset 算法选择

from typing import Callable, Dict, List
from dataclasses import dataclass

@dataclass
class DomainEvent:
    name: str
    payload: dict


class EventBus:
    """轻量级内存事件总线，用于模块间解耦通信。"""
    
    def __init__(self):
        self._listeners: Dict[str, List[Callable]] = {}
    
    def subscribe(self, event_name: str, handler: Callable):
        self._listeners.setdefault(event_name, []).append(handler)
    
    def publish(self, event: DomainEvent):
        for handler in self._listeners.get(event.name, []):
            try:
                handler(event.payload)
            except Exception as e:
                # 日志记录，不中断其他监听器
                print(f"Event handler error: {e}")
```

class TrainingService:
    """
    模型训练调度服务。
    负责：
    - 准备训练/验证/测试数据集
    - 触发 Celery 异步训练任务
    - 追踪训练进度和指标
    - 主动学习曲线分析
    """
    
    def __init__(self, trainer: BaseModelTrainer, db_session, celery_app):
        self.trainer = trainer
        self.db = db_session
        self.celery = celery_app
    
    def launch_training(self, config: TrainingConfig) -> int:
        """启动训练任务，返回任务记录ID"""
        task = TrainingTask(
            task_name=config.name,
            model_type=config.model_type,
            dataset_config=config.dataset_config,
            hyperparams=config.hyperparams,
            status="PENDING"
        )
        self.db.add(task)
        self.db.commit()
        
        # 提交 Celery 任务
        celery_job = self.celery.send_task(
            "tasks.train_model",
            args=[task.id, config.to_dict()]
        )
        task.celery_task_id = celery_job.id
        self.db.commit()
        
        return task.id
    
    def get_active_learning_curve(self) -> List[dict]:
        """
        获取主动学习效果曲线：
        x轴：已标注样本数，y轴：F1 / 等级准确率 / 素养检出率
        用于验证 Coreset 选择是否优于随机选择。
        """
        # 查询历次训练任务结果
        tasks = self.db.query(TrainingTask).filter(
            TrainingTask.status == "SUCCESS"
        ).order_by(TrainingTask.created_at).all()
        
        return [
            {
                "labeled_count": t.dataset_config.get("labeled_count", 0),
                "f1_score": t.metrics.get("f1", 0),
                "level_accuracy": t.metrics.get("level_accuracy", 0),
                "detection_rate": t.metrics.get("detection_rate", 0),
                "algorithm": t.dataset_config.get("selection_algorithm", "random")
            }
            for t in tasks
        ]
```
